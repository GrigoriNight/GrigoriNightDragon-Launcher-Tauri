         #[cfg(target_os = "android")]

         #[cfg(not(target_os = "android"))]

             let dest = app.path().app_local_data_dir().map_err(|e| e.to_string())?.join("game");
             std::fs::create_dir_all(&dest).map_err(|e| e.to_string())?;

             let mut bytes = Vec::new();

             let mut archive = zip::ZipArchive::new(std::io::Cursor::new(bytes)).map_err(|e| e.to_string())?;
             for i in 0..archive.len() {
                 let mut entry = archive.by_index(i).map_err(|e| e.to_string())?;
                 let out = match entry.enclosed_name() { Some(p) => dest.join(p), None => continue };
                 if entry.is_dir() { std::fs::create_dir_all(&out).ok(); }
                 else {
                     if let Some(parent) = out.parent() { std::fs::create_dir_all(parent).map_err(|e| e.to_string())?; }

                     std::io::copy(&mut entry, &mut w).map_err(|e| e.to_string())?;

             }
